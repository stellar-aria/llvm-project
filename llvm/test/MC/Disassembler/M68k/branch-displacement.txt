# RUN: llvm-mc -disassemble -triple m68k -mcpu=M68020 %s | FileCheck %s

## 8- and 16-bit branch displacements are signed.

# CHECK: bra -$80
0x60 0x80
# CHECK: bra -$2
0x60 0xfe
# CHECK: bra $7f
0x60 0x7f
# CHECK: bra -$8000
0x60 0x00 0x80 0x00
# CHECK: bra -$4
0x60 0x00 0xff 0xfc
# CHECK: bra $7fff
0x60 0x00 0x7f 0xff

# CHECK: beq -$80
0x67 0x80
# CHECK: beq -$2
0x67 0xfe
# CHECK: beq -$8000
0x67 0x00 0x80 0x00
# CHECK: beq -$4
0x67 0x00 0xff 0xfc

# CHECK: bsr.b -$80
0x61 0x80
# CHECK: bsr.b -$2
0x61 0xfe
# CHECK: bsr.w -$8000
0x61 0x00 0x80 0x00
# CHECK: bsr.w -$4
0x61 0x00 0xff 0xfc
