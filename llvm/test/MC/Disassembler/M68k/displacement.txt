# RUN: llvm-mc -disassemble -triple m68k -mcpu=M68000 %s | FileCheck %s
# RUN: llvm-mc -disassemble -triple m68k -mcpu=M68020 %s | FileCheck %s
# RUN: llvm-mc -disassemble -triple m68k -mcpu=M68000 %s \
# RUN:   | llvm-mc -triple m68k -mcpu=M68000 -show-encoding \
# RUN:   | FileCheck %s --check-prefix=ENC

## The 8-bit displacement of (d8,An,Xn) and (d8,PC,Xn) and the 16-bit
## displacement of (d16,An) and (d16,PC) are signed. The printed text
## assembles back to the same bytes. (The assembler always uses the full
## extension word for (d8,PC,Xn) on the 68020, so this is checked on the
## 68000 only.)

# CHECK: move.l (-128,%a0,%d0), %d0
# ENC:   encoding: [0x20,0x30,0x08,0x80]
0x20 0x30 0x08 0x80
# CHECK: move.l (-2,%a0,%d0), %d0
# ENC:   encoding: [0x20,0x30,0x08,0xfe]
0x20 0x30 0x08 0xfe
# CHECK: move.l (127,%a0,%d0), %d0
# ENC:   encoding: [0x20,0x30,0x08,0x7f]
0x20 0x30 0x08 0x7f

# CHECK: move.l (-128,%pc,%d0), %d0
# ENC:   encoding: [0x20,0x3b,0x08,0x80]
0x20 0x3b 0x08 0x80
# CHECK: move.l (-2,%pc,%d0), %d0
# ENC:   encoding: [0x20,0x3b,0x08,0xfe]
0x20 0x3b 0x08 0xfe
# CHECK: move.l (127,%pc,%d0), %d0
# ENC:   encoding: [0x20,0x3b,0x08,0x7f]
0x20 0x3b 0x08 0x7f

# CHECK: move.l (-32768,%a0), %d0
# ENC:   encoding: [0x20,0x28,0x80,0x00]
0x20 0x28 0x80 0x00
# CHECK: move.l (-2,%a0), %d0
# ENC:   encoding: [0x20,0x28,0xff,0xfe]
0x20 0x28 0xff 0xfe
# CHECK: move.l (32767,%a0), %d0
# ENC:   encoding: [0x20,0x28,0x7f,0xff]
0x20 0x28 0x7f 0xff

# CHECK: lea (-32768,%pc), %a0
# ENC:   encoding: [0x41,0xfa,0x80,0x00]
0x41 0xfa 0x80 0x00
# CHECK: lea (-2,%pc), %a0
# ENC:   encoding: [0x41,0xfa,0xff,0xfe]
0x41 0xfa 0xff 0xfe
# CHECK: lea (32767,%pc), %a0
# ENC:   encoding: [0x41,0xfa,0x7f,0xff]
0x41 0xfa 0x7f 0xff
