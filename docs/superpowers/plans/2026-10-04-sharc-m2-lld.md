# SHARC M2: Linking with LLD — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** LLD links SHARC objects for adsp-2156x and adsp-2137x, resolving word-addressed fields through a data-sheet block table; llvm-objdump symbolizes absolute addresses; digiemu's sharcemu runs linked 2156x programs.

**Architecture:** A per-CPU block table (`SHARCMemoryMap.def` in TargetParser) is the single source for byte↔word conversion, used by LLD, llvm-objdump and (by parsing the .def) digiemu. Relocation field placement moves to a header in BinaryFormat shared by the MC asm backend and LLD. LLD gains `lld/ELF/Arch/SHARC.cpp`.

**Tech Stack:** C++ (LLVM, LLD), TableGen-free `.def` tables, lit/FileCheck, gtest, yaml2obj; Python (digiemu sharcemu, pytest, uv).

**Spec:** `docs/superpowers/specs/2026-10-04-sharc-m2-lld-design.md`

#include "llvm/TargetParser/SHARCMemoryMap.h"
#include "llvm/BinaryFormat/ELF.h"
#include "gtest/gtest.h"

using namespace llvm;
using namespace llvm::SHARC;

namespace {

uint64_t word(StringRef CPU, AddrKind K, uint64_t Byte) {
  const Block *B = findBlock(CPU, Byte);
  EXPECT_NE(B, nullptr) << "no block at " << Byte;
  if (!B)
    return NoView;
  return toWord(*B, K, Byte).value_or(NoView);
}

// ADSP-2156x data sheet Rev. D, Table 2 (L1) and Table 3 (L2, 8 Mb).
TEST(SHARCMemoryMap, ADSP2156xL1Ends) {
  StringRef C = "adsp-2156x";
  // Block 0: byte 0x00240000-0x0026FFFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x240000), 0x48000u);
  EXPECT_EQ(word(C, AddrKind::LW, 0x26FFF8), 0x4DFFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x240000), 0x90000u);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x240000 + 6 * 0x7FFF), 0x97FFFu);
  EXPECT_EQ(word(C, AddrKind::NW, 0x26FFFC), 0x9BFFFu);
  EXPECT_EQ(word(C, AddrKind::SWCode, 0x26FFFE), 0x137FFFu);
  EXPECT_EQ(word(C, AddrKind::SWData, 0x240002), 0x120001u);
  EXPECT_EQ(word(C, AddrKind::Byte, 0x26FFFF), 0x26FFFFu);
  // Block 1: byte 0x002C0000-0x002EFFFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x2EFFF8), 0x5DFFFu);
  EXPECT_EQ(word(C, AddrKind::NW, 0x2EFFFC), 0xBBFFFu);
  EXPECT_EQ(word(C, AddrKind::SWCode, 0x2EFFFE), 0x177FFFu);
  // Block 2: byte 0x00300000-0x0031FFFF; ISA ends at 0xC5554.
  EXPECT_EQ(word(C, AddrKind::LW, 0x31FFF8), 0x63FFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x300000 + 6 * 0x5554), 0xC5554u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x31FFFC), 0xC7FFFu);
  EXPECT_EQ(word(C, AddrKind::SWCode, 0x31FFFE), 0x18FFFFu);
  // Block 3: byte 0x00380000-0x0039FFFF; ISA ends at 0xE5554.
  EXPECT_EQ(word(C, AddrKind::LW, 0x39FFF8), 0x73FFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x380000 + 6 * 0x5554), 0xE5554u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x39FFFC), 0xE7FFFu);
  EXPECT_EQ(word(C, AddrKind::SWCode, 0x39FFFE), 0x1CFFFFu);
}

TEST(SHARCMemoryMap, ADSP2156xL2) {
  StringRef C = "adsp-2156x";
  // L2 RAM 8 Mb: byte 0x20000000-0x200FFFFF, NW 0x08000000-0x0803FFFF,
  // VISA 0x00B80000-0x00BFFFFF, ISA from 0x005C0000.
  EXPECT_EQ(word(C, AddrKind::NW, 0x200FFFFC), 0x0803FFFFu);
  EXPECT_EQ(word(C, AddrKind::SWCode, 0x200FFFFE), 0xBFFFFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x20000000), 0x5C0000u);
  // The table prints the last ISA word as 0x5EAAAA, which would end 2 bytes
  // past L2; whole words end at 0x5EAAA9.
  EXPECT_EQ(word(C, AddrKind::ISA, 0x20000000 + 6 * 0x2AAA9), 0x5EAAA9u);
  const Block *B = findBlock(C, 0x20000000);
  ASSERT_NE(B, nullptr);
  EXPECT_FALSE(toWord(*B, AddrKind::LW, 0x20000000));
  EXPECT_FALSE(toWord(*B, AddrKind::SWData, 0x20000000));
}

// ADSP-21371/21375 data sheet Rev. D, Table 3 (RAM). Byte = 2 * SW.
TEST(SHARCMemoryMap, ADSP2137xRAMEnds) {
  StringRef C = "adsp-2137x";
  // Block 0 RAM: SW 0x130000-0x133FFF, so byte 0x260000-0x267FFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x260000), 0x4C000u);
  EXPECT_EQ(word(C, AddrKind::LW, 0x267FF8), 0x4CFFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x260000), 0x90000u);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x260000 + 6 * 0x1554), 0x91554u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x260000), 0x98000u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x267FFC), 0x99FFFu);
  EXPECT_EQ(word(C, AddrKind::SWData, 0x267FFE), 0x133FFFu);
  // Block 1 RAM: SW 0x170000-0x173FFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x2E7FF8), 0x5CFFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x2E0000), 0xB0000u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x2E7FFC), 0xB9FFFu);
  // Block 2 RAM: SW 0x180000-0x183FFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x307FF8), 0x60FFFu);
  EXPECT_EQ(word(C, AddrKind::ISA, 0x300000 + 6 * 0x1554), 0xC1554u);
  EXPECT_EQ(word(C, AddrKind::NW, 0x307FFC), 0xC1FFFu);
  // Block 3 RAM: SW 0x1C0000-0x1C3FFF.
  EXPECT_EQ(word(C, AddrKind::LW, 0x387FF8), 0x70FFFu);
  EXPECT_EQ(word(C, AddrKind::NW, 0x387FFC), 0xE1FFFu);
  EXPECT_EQ(word(C, AddrKind::SWData, 0x387FFE), 0x1C3FFFu);
  // Classic SHARC has no byte space.
  const Block *B = findBlock(C, 0x260000);
  ASSERT_NE(B, nullptr);
  EXPECT_FALSE(toWord(*B, AddrKind::Byte, 0x260000));
}

TEST(SHARCMemoryMap, Rejects) {
  StringRef C = "adsp-2156x";
  EXPECT_EQ(findBlock(C, 0x100000), nullptr);    // between blocks
  EXPECT_EQ(findBlock("adsp-214xx", 0x240000), nullptr); // not in M2
  const Block *B = findBlock(C, 0x240000);
  ASSERT_NE(B, nullptr);
  EXPECT_FALSE(toWord(*B, AddrKind::NW, 0x240002));  // misaligned
  EXPECT_FALSE(toWord(*B, AddrKind::ISA, 0x240004));
  EXPECT_FALSE(toWord(*B, AddrKind::ISA, 0x26FFFC)); // partial last row
}

TEST(SHARCMemoryMap, ToByteInvertsToWord) {
  for (StringRef C : {"adsp-2156x", "adsp-2137x"})
    for (const Block &B : blocks(C))
      for (unsigned K = 0; K < 6; ++K) {
        if (B.WordBase[K] == NoView)
          continue;
        auto Kind = AddrKind(K);
        uint64_t Last = B.ByteBase +
                        (B.Size / unitBytes(Kind) - 1) * unitBytes(Kind);
        for (uint64_t Byte : {B.ByteBase, Last}) {
          std::optional<uint64_t> W = toWord(B, Kind, Byte);
          ASSERT_TRUE(W) << B.Name << " kind " << K;
          EXPECT_EQ(toByte(C, Kind, *W), Byte) << B.Name << " kind " << K;
        }
      }
}

TEST(SHARCMemoryMap, EFlags) {
  EXPECT_EQ(memoryMapCPU(ELF::EF_SHARC_MACH_2137X), "adsp-2137x");
  EXPECT_EQ(memoryMapCPU(ELF::EF_SHARC_MACH_SHARCPLUS), "adsp-2156x");
  EXPECT_EQ(memoryMapCPU(ELF::EF_SHARC_MACH_214XX), "");
  EXPECT_EQ(memoryMapCPU(ELF::EF_SHARC_MACH_NONE), "");
}

```cpp
//===- SHARCMemoryMap.h - SHARC memory blocks per processor -----*- C++ -*-===//
//
// Part of the LLVM Project, under the Apache License v2.0 with LLVM Exceptions.
// See https://llvm.org/LICENSE.txt for license information.
// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception
//
//===----------------------------------------------------------------------===//
//
// Conversions between SHARC ELF byte addresses and the word addresses that
// instruction fields hold, per memory block (SHARCMemoryMap.def).
//
//===----------------------------------------------------------------------===//

#ifndef LLVM_TARGETPARSER_SHARCMEMORYMAP_H
#define LLVM_TARGETPARSER_SHARCMEMORYMAP_H

#include "llvm/ADT/ArrayRef.h"
#include "llvm/ADT/StringRef.h"
#include <cstdint>
#include <optional>

namespace llvm::SHARC {

/// Address kinds of instruction fields: long word, normal word, 48-bit
/// instruction (ISA code), short word as a VISA code address, short word as
/// a data address, and byte.
enum class AddrKind { LW, NW, ISA, SWCode, SWData, Byte };

inline constexpr uint64_t NoView = ~0ULL;

struct Block {
  StringRef CPU, Name;
  uint64_t ByteBase, Size;
  uint64_t WordBase[6]; ///< Indexed by AddrKind; NoView if absent.
};

/// Bytes per word of \p K.
unsigned unitBytes(AddrKind K);

/// The blocks of \p CPU; empty if the CPU has no table.
ArrayRef<Block> blocks(StringRef CPU);

/// The block of \p CPU containing \p Byte, or null.
const Block *findBlock(StringRef CPU, uint64_t Byte);

/// The word address of \p Byte in \p B, if \p B has a view of \p K, \p Byte
/// starts a whole word of it, and the word lies inside the block.
std::optional<uint64_t> toWord(const Block &B, AddrKind K, uint64_t Byte);

/// The byte address of word \p Word of kind \p K on \p CPU.
std::optional<uint64_t> toByte(StringRef CPU, AddrKind K, uint64_t Word);

/// The CPU whose table serves an object with these ELF flags, or "".
/// EF_SHARC_MACH_SHARCPLUS uses the ADSP-2156x table.
StringRef memoryMapCPU(unsigned EFlags);

} // namespace llvm::SHARC

#endif // LLVM_TARGETPARSER_SHARCMEMORYMAP_H
```

//
// Part of the LLVM Project, under the Apache License v2.0 with LLVM Exceptions.
// See https://llvm.org/LICENSE.txt for license information.
// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception
//
//===----------------------------------------------------------------------===//

#include "llvm/TargetParser/SHARCMemoryMap.h"
#include "llvm/BinaryFormat/ELF.h"

using namespace llvm;
using namespace llvm::SHARC;

static const Block Blocks[] = {
#define SHARC_BLOCK(CPU, NAME, BYTE, SIZE, LW, NW, ISA, SWC, SWD, B)           \
  {CPU, NAME, BYTE, SIZE, {LW, NW, ISA, SWC, SWD, B}},

};

unsigned SHARC::unitBytes(AddrKind K) {
  static const unsigned Units[] = {8, 4, 6, 2, 2, 1};
  return Units[unsigned(K)];
}

ArrayRef<Block> SHARC::blocks(StringRef CPU) {
  const Block *Begin = nullptr, *End = nullptr;
  for (const Block &B : Blocks)
    if (B.CPU == CPU) {
      if (!Begin)
        Begin = &B;
      End = &B + 1;
    }
  return Begin ? ArrayRef<Block>(Begin, End) : ArrayRef<Block>();
}

const Block *SHARC::findBlock(StringRef CPU, uint64_t Byte) {
  for (const Block &B : blocks(CPU))
    if (Byte >= B.ByteBase && Byte - B.ByteBase < B.Size)
      return &B;
  return nullptr;
}

std::optional<uint64_t> SHARC::toWord(const Block &B, AddrKind K,
                                      uint64_t Byte) {
  uint64_t Base = B.WordBase[unsigned(K)];
  unsigned Unit = unitBytes(K);
  if (Base == NoView || Byte < B.ByteBase)
    return std::nullopt;
  uint64_t Off = Byte - B.ByteBase;
  if (Off % Unit || Off / Unit >= B.Size / Unit)
    return std::nullopt;
  return Base + Off / Unit;
}

std::optional<uint64_t> SHARC::toByte(StringRef CPU, AddrKind K,
                                      uint64_t Word) {
  for (const Block &B : blocks(CPU)) {
    uint64_t Base = B.WordBase[unsigned(K)];
    unsigned Unit = unitBytes(K);
    if (Base != NoView && Word >= Base && Word - Base < B.Size / Unit)
      return B.ByteBase + (Word - Base) * Unit;
  }
  return std::nullopt;
}

StringRef SHARC::memoryMapCPU(unsigned EFlags) {
  switch (EFlags & ELF::EF_SHARC_MACH) {
  case ELF::EF_SHARC_MACH_2137X:
    return "adsp-2137x";
  case ELF::EF_SHARC_MACH_SHARCPLUS:
    return "adsp-2156x";
  default:
    return "";
  }
}

//===- SHARCRelocFields.h - SHARC relocation field bits ---------*- C++ -*-===//
//
// Part of the LLVM Project, under the Apache License v2.0 with LLVM Exceptions.
// See https://llvm.org/LICENSE.txt for license information.
// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception
//
//===----------------------------------------------------------------------===//
//
// The bits each R_SHARC_* relocation fills, in the 48-bit instruction window
// (llvm/docs/SHARCUsage.rst, "Relocations"), and how to write them into an
// instruction stored as 16-bit little-endian parcels, most significant parcel
// first. Shared by the MC asm backend and LLD.
//
//===----------------------------------------------------------------------===//

#ifndef LLVM_BINARYFORMAT_SHARCRELOCFIELDS_H
#define LLVM_BINARYFORMAT_SHARCRELOCFIELDS_H

#include "llvm/ADT/ArrayRef.h"
#include "llvm/BinaryFormat/ELF.h"
#include "llvm/Support/MathExtras.h"
#include <algorithm>
#include <cstdint>
#include <utility>

namespace llvm::SHARC {

/// (hi, lo) bits of a field in the 48-bit window, most significant first.
using FieldSegment = std::pair<unsigned, unsigned>;

inline ArrayRef<FieldSegment> relocFieldSegments(uint32_t Type) {
  static const FieldSegment Bits24[] = {{23, 0}};
  static const FieldSegment Bits23[] = {{22, 0}};
  static const FieldSegment Bits6[] = {{32, 32}, {31, 27}};
  static const FieldSegment Bits32[] = {{31, 0}};
  switch (Type) {
  case ELF::R_SHARC_ADDR24_NW:
  case ELF::R_SHARC_ADDR24_SW:
  case ELF::R_SHARC_PCREL24_ISA:
  case ELF::R_SHARC_PCREL24_VISA:
    return Bits24;
  case ELF::R_SHARC_PCREL23_ISA:
  case ELF::R_SHARC_PCREL23_VISA:
    return Bits23;
  case ELF::R_SHARC_PCREL6_ISA:
  case ELF::R_SHARC_PCREL6_VISA:
    return Bits6;
  case ELF::R_SHARC_ADDR32_NW:
  case ELF::R_SHARC_ADDR32_SW:
  case ELF::R_SHARC_ADDR32_LW:
  case ELF::R_SHARC_ADDR32_BYTE:
    return Bits32;
  default:
    return {};
  }
}

inline unsigned fieldWidth(ArrayRef<FieldSegment> Segs) {
  unsigned W = 0;
  for (auto [Hi, Lo] : Segs)
    W += Hi - Lo + 1;
  return W;
}

inline bool isPCRelReloc(uint32_t Type) {
  return Type >= ELF::R_SHARC_PCREL24_ISA && Type <= ELF::R_SHARC_PCREL6_VISA;
}

/// Bytes per PC-relative unit: a 48-bit instruction (ISA) or a 16-bit
/// parcel (VISA).
inline unsigned pcRelRelocUnit(uint32_t Type) {
  return (Type - ELF::R_SHARC_PCREL24_ISA) % 2 ? 2 : 6;
}

/// Scatter the low fieldWidth(Segs) bits of \p V over \p Segs of the
/// instruction at \p Insn, rewriting only the parcels the field touches.
inline void insertField(uint8_t *Insn, ArrayRef<FieldSegment> Segs,
                        uint64_t V) {
  unsigned Remaining = fieldWidth(Segs);
  for (auto [Hi, Lo] : Segs) {
    unsigned N = Hi - Lo + 1;
    Remaining -= N;
    uint64_t Part = (V >> Remaining) & maskTrailingOnes<uint64_t>(N);
    for (unsigned P = (47 - Hi) / 16; P <= (47 - Lo) / 16; ++P) {
      unsigned PLo = 32 - 16 * P; // window bit of the parcel's bit 0
      uint16_t Parcel = Insn[2 * P] | (uint16_t(Insn[2 * P + 1]) << 8);
      unsigned From = std::max(Lo, PLo), To = std::min(Hi, PLo + 15);
      uint16_t Mask = maskTrailingOnes<uint16_t>(To - From + 1) << (From - PLo);
      uint16_t Bits = uint16_t((Part >> (From - Lo)) << (From - PLo)) & Mask;
      Parcel = (Parcel & ~Mask) | Bits;
      Insn[2 * P] = uint8_t(Parcel);
      Insn[2 * P + 1] = uint8_t(Parcel >> 8);
    }
  }
}

} // namespace llvm::SHARC

#endif // LLVM_BINARYFORMAT_SHARCRELOCFIELDS_H

//===- SHARC.cpp ----------------------------------------------------------===//
//
// Part of the LLVM Project, under the Apache License v2.0 with LLVM Exceptions.
// See https://llvm.org/LICENSE.txt for license information.
// SPDX-License-Identifier: Apache-2.0 WITH LLVM-exception
//
//===----------------------------------------------------------------------===//
//
// Analog Devices SHARC and SHARC+ (llvm/docs/SHARCUsage.rst). ELF addresses
// are byte addresses; instruction fields hold word addresses whose meaning
// depends on the memory block, converted with the per-processor table in
// llvm/TargetParser/SHARCMemoryMap.def.
//
//===----------------------------------------------------------------------===//

#include "InputFiles.h"
#include "Symbols.h"
#include "Target.h"
#include "llvm/BinaryFormat/ELF.h"
#include "llvm/BinaryFormat/SHARCRelocFields.h"
#include "llvm/Support/Endian.h"
#include "llvm/TargetParser/SHARCMemoryMap.h"

using namespace llvm;
using namespace llvm::object;
using namespace llvm::support::endian;
using namespace llvm::ELF;
using namespace lld;
using namespace lld::elf;

namespace {
class SHARC final : public TargetInfo {
public:
  SHARC(Ctx &);
  uint32_t calcEFlags() const override;
  RelExpr getRelExpr(RelType type, const Symbol &s,
                     const uint8_t *loc) const override;
  void relocate(uint8_t *loc, const Relocation &rel,
                uint64_t val) const override;

private:
  // The processor whose memory table resolves word addresses; set by
  // calcEFlags, which runs before any relocation is applied.
  mutable StringRef cpu;
};
} // namespace

SHARC::SHARC(Ctx &ctx) : TargetInfo(ctx) {
  // IDLE is not a trap; a zero word decodes as NOP (Type 21a).
  trapInstr = {0, 0, 0, 0};
}

static uint32_t getEFlags(InputFile *file) {
  return cast<ObjFile<ELF32LE>>(file)->getObj().getHeader().e_flags;
}

uint32_t SHARC::calcEFlags() const {
  uint32_t mach = EF_SHARC_MACH_NONE;
  InputFile *first = nullptr;
  for (InputFile *f : ctx.objectFiles) {
    uint32_t m = getEFlags(f) & EF_SHARC_MACH;
    if (m == EF_SHARC_MACH_NONE)
      continue;
    if (mach == EF_SHARC_MACH_NONE) {
      mach = m;
      first = f;
    } else if (m != mach) {
      ErrAlways(ctx) << f << ": cannot link object files for different SHARC "
                     << "processors (" << first << " has machine " << mach
                     << ", this file " << m << ")";
    }
  }
  cpu = SHARC::memoryMapCPU(mach);
  if (cpu.empty())
    ErrAlways(ctx) << "no SHARC memory map for machine " << mach
                   << " (e_flags); M2 supports adsp-2137x and adsp-2156x";
  return mach;
}

RelExpr SHARC::getRelExpr(RelType type, const Symbol &s,
                          const uint8_t *loc) const {
  switch (type) {
  case R_SHARC_NONE:
    return R_NONE;
  case R_SHARC_PCREL24_ISA:
  case R_SHARC_PCREL24_VISA:
  case R_SHARC_PCREL23_ISA:
  case R_SHARC_PCREL23_VISA:
  case R_SHARC_PCREL6_ISA:
  case R_SHARC_PCREL6_VISA:
    return R_PC;
  case R_SHARC_DATA32:
  case R_SHARC_ADDR24_NW:
  case R_SHARC_ADDR24_SW:
  case R_SHARC_ADDR32_NW:
  case R_SHARC_ADDR32_SW:
  case R_SHARC_ADDR32_LW:
  case R_SHARC_ADDR32_BYTE:
    return R_ABS;
  default:
    Err(ctx) << getErrorLoc(ctx, loc) << "unsupported relocation " << type
             << " against symbol " << &s;
    return R_NONE;
  }
}

void SHARC::relocate(uint8_t *loc, const Relocation &rel, uint64_t val) const {
  if (rel.type == R_SHARC_DATA32) {
    checkIntUInt(ctx, loc, val, 32, rel);
    write32le(loc, val);
    return;
  }
  auto segs = SHARC::relocFieldSegments(rel.type);
  unsigned width = SHARC::fieldWidth(segs);
  if (SHARC::isPCRelReloc(rel.type)) {
    // val = S + A - P in bytes; P is the instruction's first byte.
    int64_t off = int64_t(val);
    unsigned unit = SHARC::pcRelRelocUnit(rel.type);
    if (off % unit) {
      Err(ctx) << getErrorLoc(ctx, loc) << "branch target of " << rel.type
               << " against " << rel.sym << " is not on an instruction boundary";
      return;
    }
    checkInt(ctx, loc, off / int64_t(unit), width, rel);
    SHARC::insertField(loc, segs, uint64_t(off / int64_t(unit)));
    return;
  }
  Err(ctx) << getErrorLoc(ctx, loc) << "unrecognized relocation " << rel.type;
}

void elf::setSHARCTargetInfo(Ctx &ctx) { ctx.target.reset(new SHARC(ctx)); }

void SHARC::checkOutputSections() const {

    if (!(osec->flags & SHF_EXECINSTR))
      continue;

namespace {
/// Absolute addresses in instruction fields are word addresses; map them
/// back to ELF byte addresses so llvm-objdump can name their symbols.
class SHARCMCInstrAnalysis : public MCInstrAnalysis {
public:
  explicit SHARCMCInstrAnalysis(const MCInstrInfo *Info)
      : MCInstrAnalysis(Info) {}

  std::optional<uint64_t>
  evaluateMemoryOperandAddress(const MCInst &Inst, const MCSubtargetInfo *STI,
                               uint64_t Addr, uint64_t Size) const override {
    if (!STI)
      return std::nullopt;
    StringRef CPU = STI->getCPU();
    bool VISA = STI->hasFeature(SHARC::FeatureVISASection);
    std::optional<SHARC::AddrKind> Kind;
    unsigned Op = 0;
    switch (Inst.getOpcode()) {
    case SHARC::T8a_abs:
    case SHARC::T25a_direct:
      Kind = VISA ? SHARC::AddrKind::SWCode : SHARC::AddrKind::ISA;
      Op = Inst.getNumOperands() - 1; // addr is the last operand
      break;
    case SHARC::T14a:
      // l selects long-word access (PGR Type 14a).
      Kind = Inst.getOperand(2).getImm() ? SHARC::AddrKind::LW
                                         : SHARC::AddrKind::NW;
      Op = Inst.getNumOperands() - 1;
      break;
    default:
      return std::nullopt;
    }
    if (CPU == "adsp-sc58x" || CPU == "adsp-2158x")
      CPU = "adsp-2156x"; // same map until those CPUs get rows
    return SHARC::toByte(CPU, *Kind, Inst.getOperand(Op).getImm());
  }
};
} // namespace

static MCInstrAnalysis *createSHARCMCInstrAnalysis(const MCInstrInfo *Info) {
  return new SHARCMCInstrAnalysis(Info);
}

```python
"""The SHARC memory block table, read from LLVM's SHARCMemoryMap.def so the
emulator and the linker share one source (LLVM M2 spec, section 3)."""
import os
import re
from functools import lru_cache
from pathlib import Path

DEF = Path(os.environ.get("SHARC_LLVM_ROOT", Path.home() / "GitHub/llvm-project")) / \
    "llvm/include/llvm/TargetParser/SHARCMemoryMap.def"
KINDS = ("lw", "nw", "isa", "sw_code", "sw_data", "byte")
UNITS = {"lw": 8, "nw": 4, "isa": 6, "sw_code": 2, "sw_data": 2, "byte": 1}
_ROW = re.compile(r'^SHARC_BLOCK\("([^"]+)",\s*"([^"]+)",(.*)\)\s*$')


@lru_cache(maxsize=None)
def blocks(cpu):
    out = []
    for line in DEF.read_text().splitlines():
        m = _ROW.match(line)
        if not m or m[1] != cpu:
            continue
        vals = [None if v.strip() == "NONE" else int(v, 0) for v in m[3].split(",")]
        base, size, *words = vals
        out.append({"name": m[2], "base": base, "size": size, **dict(zip(KINDS, words))})
    return out


def find_block(cpu, byte):
    return next((b for b in blocks(cpu) if b["base"] <= byte < b["base"] + b["size"]), None)


def to_word(cpu, kind, byte):
    b = find_block(cpu, byte)
    if b is None or b[kind] is None:
        return None
    off, unit = byte - b["base"], UNITS[kind]
    if off % unit or off // unit >= b["size"] // unit:
        return None
    return b[kind] + off // unit
```

```python
"""Load an LLD-linked SHARC+ executable into a sharcemu Image (LLVM M2 spec,
section 7). Segments go to their byte addresses through the emulator's
loader mapping, so data and code fetches see them."""
import struct

from . import memmap
from .machine import loader_address

EM_SHARC = 133
EF_SHARC_MACH = 0xF
EF_SHARC_MACH_SHARCPLUS = 4


def _symbols(data, shoff, shentsize, shnum):
    secs = [struct.unpack_from("<IIIIIIIIII", data, shoff + i * shentsize) for i in range(shnum)]
    syms = {}
    for s in secs:
        if s[1] != 2:  # SHT_SYMTAB
            continue
        strtab = secs[s[6]]
        for off in range(s[4], s[4] + s[5], 16):
            name, value, _size, _info, _other, _shndx = struct.unpack_from("<IIIBBH", data, off)
            end = data.index(b"\0", strtab[4] + name)
            n = data[strtab[4] + name:end].decode()
            if n:
                syms[n] = value
    return syms


def load(path, image, cpu="adsp-2156x"):
    data = open(path, "rb").read()
    if data[:4] != b"\x7fELF" or data[4] != 1 or data[5] != 1:
        raise ValueError(f"{path}: not a 32-bit little-endian ELF file")
    (_type, machine, _ver, entry, phoff, shoff, flags, _ehsize, phentsize, phnum,
     shentsize, shnum, _shstrndx) = struct.unpack_from("<HHIIIIIHHHHHH", data, 16)
    if machine != EM_SHARC or flags & EF_SHARC_MACH != EF_SHARC_MACH_SHARCPLUS:
        raise ValueError(f"{path}: not an ADSP-2156x SHARC executable "
                         f"(machine {machine}, e_flags {flags:#x})")
    for i in range(phnum):
        p_type, p_offset, _vaddr, p_paddr, p_filesz, p_memsz, _f, _a = \
            struct.unpack_from("<IIIIIIII", data, phoff + i * phentsize)
        if p_type != 1:  # PT_LOAD
            continue
        image.put(loader_address(p_paddr), data[p_offset:p_offset + p_filesz])
        if p_memsz > p_filesz:
            image.put(loader_address(p_paddr + p_filesz), bytes(p_memsz - p_filesz))
    symbols = _symbols(data, shoff, shentsize, shnum)
    entry_sw = memmap.to_word(cpu, "sw_code", entry)
    if entry_sw is None:
        raise ValueError(f"{path}: entry {entry:#x} is not VISA code memory")
    return {"entry_sw": entry_sw, "symbols": symbols}
```
